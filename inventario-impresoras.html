function doGet(e) {
  try {
    var sheet = SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName("Impresoras");
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["ID","Fecha","Hora","Marca","Modelo",
        "Ubicacion","Estado","Tipo","Responsable","Observaciones","Dispositivo"]);
    }
    var p = e.parameter;
    sheet.appendRow([p.id||'',p.fecha||'',p.hora||'',p.marca||'',
      p.modelo||'',p.ubicacion||'',p.estado||'',p.tipo||'',
      p.responsable||'',p.obs||'',p.device||'App']);
    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,error:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
